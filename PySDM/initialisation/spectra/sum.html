14class Sum:
15    def __init__(self, spectra: tuple, interpolation_grid=None):
16        interpolation_grid = interpolation_grid or default_interpolation_grid
17        self.spectra = spectra
18        self.norm_factor = sum((s.norm_factor for s in self.spectra))
19        percentiles = [s.percentiles(interpolation_grid) for s in self.spectra]
20        cdf_arg = np.zeros(len(interpolation_grid) * len(self.spectra) + 1)
21        cdf_arg[1:] = np.concatenate(percentiles)
22        cdf = self.cumulative(cdf_arg) / self.norm_factor
23        self.inverse_cdf = interp1d(cdf, cdf_arg, bounds_error=False)
24
25    def size_distribution(self, arg):
26        result = 0.0
27        for spectrum in self.spectra:
28            result += spectrum.size_distribution(arg)
29        return result
30
31    def cumulative(self, arg):
32        result = 0.0
33        for spectrum in self.spectra:
34            result += spectrum.cumulative(arg)
35        return result
36
37    def percentiles(self, cdf_values):
38        return self.inverse_cdf(cdf_values)
39
40    def pdf(self, arg):
41        return self.size_distribution(arg) / self.norm_factor
42
43    def cdf(self, arg):
44        return self.cumulative(arg) / self.norm_factor
